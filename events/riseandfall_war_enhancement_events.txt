# Event: Decide the fate of the previous liege after victory.
riseandfall_war_enhancement.2 = {
	type = character_event
	
	title = riseandfall_war_enhancement.2.t
	desc = riseandfall_war_enhancement.2.d

	theme = intrigue
	left_portrait = {
		character = root
		animation = admiration
	}
	right_portrait = {
		character = scope:riseandfall_war_enhancement_prev_liege
		animation = beg
	}
	
	# Option: Kill every member of their house you can find! Their house must learn.
	option = { 	
		name = riseandfall_war_enhancement.2.e
		trigger = { 
			AND = {
				OR = { 
					has_trait = sadistic
					has_trait = lunatic
					has_trait = possessed
					ai_vengefulness >= 95
					ai_compassion <= -90
					ai_honor <= -90
					has_government = tribal_government
					has_government = nomad_government
					has_government = celestial_government
					has_government = meritocratic_government
					has_government = steppe_admin_government
					has_government = administrative_government
				}
				scope:riseandfall_war_enhancement_prev_liege = {
					NOT = { house = scope:riseandfall_war_enhancement_attacker.house }
					any_close_or_extended_family_member = {
							OR = {
								is_at_location = scope:riseandfall_liege_death_location
								is_in_the_same_court_as_or_guest = scope:riseandfall_war_enhancement_attacker
								is_at_same_location = scope:riseandfall_war_enhancement_attacker
								is_vassal_or_below_of = scope:riseandfall_war_enhancement_attacker
								location.duchy = scope:riseandfall_war_enhancement_liege_capital_location.duchy
							}
							NOR = { 
								is_spouse_of = root					
							}					
					}
				}
			}
		}
		war_enhancement_execute_liege_family = yes
		war_enhancement_liege_family_slain = yes
		ai_chance = {
			base = -50
			ai_value_modifier = {
				ai_vengefulness = 0.5
				ai_boldness = 0.3
				ai_energy = 0.4
				ai_compassion = -0.6
				ai_honor = -0.4
			}
			opinion_modifier = { # Opinion Factor
				who = root
				opinion_target = scope:riseandfall_war_enhancement_prev_liege
				multiplier = -0.3
			}
			opinion_modifier = { # Opinion Factor
				who = root
				opinion_target = scope:riseandfall_war_enhancement_liege_primary_title.holder
				multiplier = -0.3
			}
			modifier = { #  we have a more savage form of government
				add = 125
				root = {
					OR = {
						has_government = nomad_government
					}
				}
			}
			modifier = { #  we have a more savage form of government
				add = 25
				root = {
					OR = {
						has_government = tribal_government
					}
				}
			}
			modifier = { #  human resources are plentiful
				add = 25
				root = {
					OR = {
						has_government = celestial_government
						has_government = meritocratic_government
						has_government = steppe_admin_government
						has_government = japan_administrative_government
						has_government = japan_fuedal_government
					}
				}
			}
		}		
	}
	
	# Option: Imprison every member of their house you can find!
	option = { 	
		name = riseandfall_war_enhancement.2.g
		trigger = { 
			AND = {
				OR = { 
					has_trait = sadistic
					has_trait = lunatic
					has_trait = possessed
					ai_vengefulness >= 50
					ai_compassion <= -50
					ai_honor <= -30
					has_government = tribal_government
					has_government = nomad_government
					has_government = celestial_government
					has_government = meritocratic_government
					has_government = steppe_admin_government
					has_government = administrative_government
				}
				scope:riseandfall_war_enhancement_prev_liege = {
					NOT = { house = scope:riseandfall_war_enhancement_attacker.house }
					any_close_or_extended_family_member = {
							OR = {
								is_at_location = scope:riseandfall_liege_death_location
								is_in_the_same_court_as_or_guest = scope:riseandfall_war_enhancement_attacker
								is_at_same_location = scope:riseandfall_war_enhancement_attacker
								is_vassal_or_below_of = scope:riseandfall_war_enhancement_attacker
								location.duchy = scope:riseandfall_war_enhancement_liege_capital_location.duchy
							}
							NOR = { 
								is_spouse_of = root
								is_imprisoned = yes
							}						
					}
				}
			}
		}
		war_enhancement_imprison_liege_family = yes
		ai_chance = {
			base = 80
			ai_value_modifier = {
				ai_vengefulness = -0.1
				ai_energy = 0.3

riseandfall_war_enhancement.23 = { # We can usurp the emperor or do other stuff
	title = riseandfall_war_enhancement.23.t
	desc = riseandfall_war_enhancement.23.d
	type = character_event
	theme = realm
	
	left_portrait = {
		character = root
		animation = admiration
	}
	right_portrait = {
		character = scope:riseandfall_war_enhancement_liege_primary_title.holder
		animation = war_over_win
	}
	
	# Option: Seize power from the emperor.
	option = { # Now's my chance, usurp the administrative ruler.
		name = riseandfall_war_enhancement.23.a
		trigger = {
			scope:riseandfall_war_enhancement_liege_primary_title.holder = {
				OR = {
					has_government = administrative_government
					is_independent_ruler = yes
				}
			}
			root = {
				is_independent_ruler = no
			}
		}
		war_enhancement_administrative_liege_usurper = yes
		war_enhancement_overthrow_emperor_administrative = yes
		scope:riseandfall_war_enhancement_prev_liege = {
			trigger_event = {
					id = riseandfall_war_enhancement.5 #everything is falling apart! (admin edition)
				}
		}
		ai_chance = {
			base = -50
			ai_value_modifier = {
				ai_greed = 2
				ai_honor = -1.5
				ai_energy = 2.5
				ai_compassion = -2
				ai_boldness = 2.5
				ai_sociability = 0.5
				ai_vengefulness = 0.3
			}
			modifier = {
				add = 20
				root = {
					has_trait = ambitious
				}
			}
			opinion_modifier = { # Opinion Factor
				who = root
				opinion_target = scope:riseandfall_war_enhancement_liege_primary_title.holder
				multiplier = -5
			}
			modifier = {
				add = 30
				root = {
					has_trait = ambitious
				}
			}
			modifier = {
				add = 75
				scope:riseandfall_war_enhancement_liege_primary_title.holder = {
					is_imprisoned_by = scope:riseandfall_war_enhancement_attacker
				}
			}
		}
	}
	# Option: Decline to take action.
	option = { # decline
		name = riseandfall_war_enhancement.23.b
		war_enhancement_enthroned_heir = yes
		scope:riseandfall_war_enhancement_prev_liege = {
			trigger_event = {
					id = riseandfall_war_enhancement.5 #everything is falling apart!
				}
		}
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_greed = -0.75
				ai_boldness = -2.5
				ai_vengefulness = -1
				ai_honor = 0.3
				ai_compassion = 0.3
				ai_rationality = -0.2
				ai_sociability = -1
			}
			modifier = {
				add = 50
				root = {
					has_trait = content
				}
			}
			
			modifier = {
				add = -50
				root = {
					has_trait = ambitious
				}
			}
			opinion_modifier = { # Opinion Factor
				who = root
				opinion_target = scope:riseandfall_war_enhancement_liege_primary_title.holder
				multiplier = 2
			}
		}
	}
}
