# This file contains on_actions that trigger scripted effects for war enhancement mechanics in the Rise and Fall mod.
# It handles events related to character deaths and war victories to simulate rebellion and succession dynamics.

on_war_won_attacker = {
	on_actions = {
		riseandfall_war_enhancement_war_won
	}
}

on_death = {
	on_actions = {
		#delay = { days = 1 }
		riseandfall_war_enhancement_regent_displaced
	}
}

riseandfall_war_enhancement_regent_displaced = {
	effect = {
		if = {
			limit = {
				root = {
					has_character_flag = council_regent_flag
				}
			}
			root.liege = {
				trigger_event = {
					id = riseandfall_war_enhancement.25
					#days = 1
					#delayed = yes
				}
			}
		}
	}
}

# Scripted effect triggered when the attacker wins a war.
# Handles post-war succession and rebellion mechanics for specific casus belli.
riseandfall_war_enhancement_war_won = {		
	effect = {
		if = {
			limit = {
				scope:war = {
					OR = {
						using_cb = liberty_faction_war
						using_cb = refused_liege_demand_war
						using_cb = claimant_faction_war
					}
				}
			}
			# Save defender as previous liege
			scope:defender = {
				save_scope_as = riseandfall_war_enhancement_prev_liege
			}
			
			# Save attacker's liege
			scope:attacker.liege = {
				save_scope_as = riseandfall_war_enhancement_liege
			}
			# Save attacker as rebellion leader
			scope:attacker = {
				save_scope_as = riseandfall_war_enhancement_rebellion_leader
			}
			
			scope:defender.capital_county = {
				save_scope_as = riseandfall_war_enhancement_liege_capital_location
			}
			
			# Find the strongest contender among attackers
			scope:war = {
				ordered_war_attacker = {
					limit = {
						is_vassal_of = scope:riseandfall_war_enhancement_liege 
					}
					order_by = { 
						add = current_military_strength
						add = {
							value = diplomacy
							multiply = 35
						}
						add = {
							value = martial
							multiply = 50
						}
						add = {
							value = stewardship
							multiply = 20
						}
						add = {
							value = intrigue
							multiply = 20
						}
						add = {
							value = learning
							multiply = 20
						}
					}
					save_scope_as = riseandfall_war_enhancement_contender
				}
			}
			
			# Determine if contender or rebellion leader becomes the attacker
			if = {
				limit = {
					OR = {
						#strongest military contender
						AND = {
							scope:riseandfall_war_enhancement_contender = { 
								current_military_strength > scope:riseandfall_war_enhancement_rebellion_leader.current_military_strength
							}
							scope:war = { 
								war_contribution = {
									target = scope:riseandfall_war_enhancement_contender 
									value > 40%
								}
							}
						}
						#diplomatic option
						AND = {
							scope:riseandfall_war_enhancement_contender = { 
								diplomacy > scope:riseandfall_war_enhancement_rebellion_leader.diplomacy
								martial > scope:riseandfall_war_enhancement_rebellion_leader.martial
								current_military_strength > scope:riseandfall_war_enhancement_rebellion_leader.current_military_strength
							}
							scope:war = { 
								war_contribution = {
									target = scope:riseandfall_war_enhancement_contender 
									value > 25%
								}
							}
						}
					}
				}
				# Set contender as attacker
				scope:riseandfall_war_enhancement_contender = {
					clear_saved_scope = riseandfall_war_enhancement_contender
					save_scope_as = riseandfall_war_enhancement_attacker
				}
			}
			else = {
				# Set rebellion leader as attacker
				scope:riseandfall_war_enhancement_rebellion_leader = {
					clear_saved_scope = riseandfall_war_enhancement_rebellion_leader
					save_scope_as = riseandfall_war_enhancement_attacker
				}
			}
			
			# Save defender's primary title
			if = {
				limit = {
					scope:war = {
						OR = {
							using_cb = liberty_faction_war
							using_cb = refused_liege_demand_war
							using_cb = claimant_faction_war
						}
					}
					scope:riseandfall_war_enhancement_attacker.primary_title = {
						target_is_de_jure_liege_or_above = scope:defender.primary_title
					}
				}
				scope:defender.primary_title = {
					save_scope_as = riseandfall_war_enhancement_liege_primary_title
				}
			}
			else_if = {
				limit = {
					scope:war = {
						OR = {
							using_cb = liberty_faction_war
							using_cb = refused_liege_demand_war
							using_cb = claimant_faction_war
						}
					}
					scope:riseandfall_war_enhancement_attacker.primary_title = {
						de_jure_liege.de_jure_liege = {
							holder = scope:defender
						}
					}
				}
				scope:riseandfall_war_enhancement_attacker.primary_title.de_jure_liege.de_jure_liege = {
					save_scope_as = riseandfall_war_enhancement_liege_primary_title
				}
			}
			else_if = {
				limit = {
					scope:war = {
						OR = {
							using_cb = liberty_faction_war
							using_cb = refused_liege_demand_war
							using_cb = claimant_faction_war
						}
					}
					scope:riseandfall_war_enhancement_attacker.primary_title = {
						de_jure_liege = {
							holder = scope:defender
						}
					}
				}
				scope:riseandfall_war_enhancement_attacker.primary_title.de_jure_liege = {
					save_scope_as = riseandfall_war_enhancement_liege_primary_title
				}
			}
			
			else_if = {
				limit = {
					scope:war = {
						OR = {
							using_cb = liberty_faction_war
							using_cb = refused_liege_demand_war
							using_cb = claimant_faction_war
						}
					}
				}
				scope:riseandfall_war_enhancement_attacker.liege.primary_title = {
					save_scope_as = riseandfall_war_enhancement_liege_primary_title
				}
			}
			
				
			
			if = {
				limit = {
					scope:riseandfall_war_enhancement_attacker.liege = {
						has_government = administrative_government
					}       
				}
				scope:war = {
					every_war_attacker = {
						limit = {
							OR = {
								is_vassal_or_below_of = scope:riseandfall_war_enhancement_attacker.liege.primary_title.holder
								has_government = landless_adventurer_government
							}
							
						}
						scope:riseandfall_war_enhancement_attacker.liege.primary_title = {
							change_appointment_investment = {
									target = prev
									value = {
										value = 10
										add = {
											value = prev.diplomacy
											multiply = 4
										}
										add = {
											value = prev.martial
											multiply = 8										
										}
										add = {
											value = prev.stewardship
											multiply = 3
										}
										add = {
											value = prev.learning
											multiply = 2
										}
										add = {
											value = prev.intrigue
											multiply = 4
										}
										add = {
											value = prev.current_military_strength
											divide = 2000
										}
									}
							}
						}
					}
				}
			}
			
			if = {
				limit = {
					scope:war = { 
							using_cb = claimant_faction_war 
					}
					scope:defender = {
						highest_held_title_tier < scope:attacker.primary_title.tier
					}
					scope:riseandfall_war_enhancement_attacker = {
						NOT = { has_government = administrative_government }
					}
				}
				# Trigger event for the attacker to choose what to do to the fallen ruler
				scope:riseandfall_war_enhancement_attacker = {
					trigger_event = {
						id = riseandfall_war_enhancement.2 # 
					}
				}
			}
			else_if = {
				limit = {
					scope:war = {
						OR = {
							using_cb = liberty_faction_war
							using_cb = refused_liege_demand_war
						}
					}
					scope:defender = {
						NOT = { has_government = administrative_government }
					}
				}
				scope:defender = {
					trigger_event = {
						id = riseandfall_war_enhancement.5 #everything is falling apart!
					}
				}
			}
			else_if = {
				limit = {
					scope:war = {
						OR = {
							using_cb = refused_liege_demand_war
							using_cb = liberty_war
						}
					}
					scope:riseandfall_war_enhancement_attacker = {
						has_government = administrative_government
					}
					scope:riseandfall_war_enhancement_liege_primary_title = {
						NOT = { current_heir = scope:riseandfall_war_enhancement_attacker }
					}
				}
				scope:riseandfall_war_enhancement_attacker = {
					trigger_event = {
						id = riseandfall_war_enhancement.23
					}
				}
			}
			else_if = {
				limit = {
					scope:war = {
						OR = {
							using_cb = refused_liege_demand_war
							using_cb = liberty_war
						}
					}
					scope:riseandfall_war_enhancement_attacker = {
						has_government = administrative_government
					}
					scope:riseandfall_war_enhancement_liege_primary_title = {
						current_heir = scope:riseandfall_war_enhancement_attacker
					}
				}
				scope:riseandfall_war_enhancement_attacker = {
					trigger_event = {
						id = riseandfall_war_enhancement.24
					}
				}
			}
		}
	}
}
