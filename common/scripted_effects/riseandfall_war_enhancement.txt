# Event .1 immediate: Imprison family members related to the killer
war_enhancement_imprison_family = {
	scope:riseandfall_war_enhancement_prev_liege = {
		every_close_or_extended_family_member = {
			limit = {
				is_independent_ruler = no
				primary_spouse = {
					OR = {
						is_at_same_location = scope:riseandfall_war_enhancement_prev_liege
						is_at_same_location = scope:riseandfall_liege_death_location
					}
					NOT = { house = scope:riseandfall_war_enhancement_attacker.house }
				}
						killer = prev
					}
				}
			}
			save_temporary_scope_as = riseandfall_war_enhancement_tbi
			root = {
				imprison = { target = scope:riseandfall_war_enhancement_tbi }
			}
		}

# Event .1.a effect: Press claims
war_enhancement_press_claims = {

	scope:riseandfall_war_enhancement_liege_primary_title.holder = {
		save_scope_as = riseandfall_war_enhancement_prev_liege_successor
	}
	
	root = {
		create_title_and_vassal_change = {
			type = conquest_claim
			save_scope_as = change
		}		
	}
	
	scope:riseandfall_war_enhancement_liege_primary_title.holder = {
		every_held_title = {
			limit = {
				root = {
					has_claim_on = prev
				}
			}
			change_title_holder_include_vassals = {
				holder = root
				change = scope:change
			}
		}
	}

	root = {
		resolve_title_and_vassal_change = scope:change
	}
	
	root = {
		create_title_and_vassal_change = {
			type = conquest_claim
			save_scope_as = change
		}		
	}
	scope:riseandfall_war_enhancement_prev_liege_successor = {
		if = { 
			limit = {
				highest_held_title_tier < scope:riseandfall_war_enhancement_attacker.primary_title.tier
			}
			change_liege = {
				liege = root
				change = scope:change
			}
		}
	}
	
	root = {
		resolve_title_and_vassal_change = scope:change
	}

	
	if = {
		limit = {
			scope:riseandfall_war_enhancement_liege_primary_title.holder != root
		}
		hidden_effect = {
			root = {
				trigger_event = {
					id = riseandfall_war_enhancement.1
				}
			}
		}
	}
	
}

# Event .6.a effect: Abdicate to heir
war_enhancement_abdicate_to_heir = {
	root = {
		create_title_and_vassal_change = {
			type = faction_demand
			save_scope_as = change
		}
		remove_character_flag = council_regent_flag
	}
	scope:riseandfall_liege_elective_title = {
		change_title_holder_include_vassals = {
			holder = scope:riseandfall_liege_elective_title.current_heir
			change = scope:change
		}
	}
	scope:riseandfall_liege_elective_title.holder = {
		every_held_title = { 
			limit = { 
				OR = {
					AND = {
						holder = scope:riseandfall_liege_elective_title.holder
						AND = {
							previous_holder = { exists = yes }
							previous_holder = scope:riseandfall_war_enhancement_prev_liege
						}
						title_held_years < 1
						OR = {
							has_claim = riseandfall_liege_elective_title.current_heir
							de_jure_liege = scope:riseandfall_liege_elective_title
							de_facto_liege = scope:riseandfall_liege_elective_title.holder
						}
					}
					this = title_capital_county
				}
			}
			change_title_holder_include_vassals = {
				holder = scope:riseandfall_liege_elective_title.current_heir
				change = scope:change
			}
		}
		every_vassal = {
			limit = {
				NOT = { this = root }
				primary_title = {
					de_jure_liege = scope:riseandfall_liege_elective_title
				}
			}				
			change_liege = {
				liege = scope:riseandfall_liege_elective_title.current_heir
				change = scope:change
			}
		}
		change_liege = {
			liege = scope:riseandfall_liege_elective_title.current_heir
			change = scope:change	
		}
		end_diarchy = yes
	}
	
	if = {
		limit = {
			scope:riseandfall_liege_elective_title.holder = {
				highest_held_title_tier < scope:riseandfall_liege_elective_title.current_heir.primary_title.tier
			}
		}
		scope:riseandfall_war_enhancement_attacker.liege = {
			change_liege = {
				liege = scope:riseandfall_liege_elective_title.current_heir
				change = scope:change
			}
		}
	}
			
	root = {
		resolve_title_and_vassal_change = scope:change
	}
	if = {
		limit = {
				NOT = { 
					scope:riseandfall_liege_elective_title = { has_title_law = princely_elective_succession_law }
				}
		}
		scope:riseandfall_liege_elective_title = {
			clear_title_laws = yes
		}
	}
}
